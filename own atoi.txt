                  }
          }
          if(c==1)
                  printf("%f\n",my_atof(argc[1]));
          else
                  printf("%d\n",my_atoi(argc[1]));

  }
 
  int my_atoi(char s[])
  {
          int i=0,add=0,sign=1;
          if(s[i]=='-')
          {
                  sign=-1;
                  i=1;
          }
          for(;s[i];i++)
          {
                  if(s[i]>='0' && s[i]<='9')
                          add=add*10+(s[i]-'0');
                  else
                          break;
          }
          return sign*add;
  }
